// Exponential Backoff & Retry Handler (OpenRouterClient.java)
public String executeWithRetry(PromptMessage prompt, int maxRetries) {
    long backoffMs = 4000; // Initial 4s delay
    for (int i = 0; i < maxRetries; i++) {
        try {
            enforceRateLimitPacing(1500); // 1.5s inter-prompt delay
            return openAiClient.chat().completions().create(buildParams(prompt));
        } catch (OpenAiException e) {
            Thread.sleep(backoffMs);
            backoffMs = Math.min(64000, backoffMs * 2); // Exponential backoff capped at 64s
        }
    }
    return fallbackStatEngine.generateDeterministicAction(prompt);
}

public Player selectJuryThreatTarget(Player voter, List<Player> activeContestants) {
    if (activeContestants.size() > 5 || voter.getSocialIntelligence() < 70) {
        return selectStandardRivalTarget(voter, activeContestants);
    }
    
    // Calculate projected jury win probabilities
    return activeContestants.stream()
        .filter(p -> !p.getId().equals(voter.getId()))
        .max(Comparator.comparingDouble(p -> calculateJuryWinProbability(p, gameContext)))
        .orElse(null);
}

// Index-Based Delta Log Streamer (GameLogService.java)
public class GameLogService {
    private final List<String> logBuffer = new CopyOnWriteArrayList<>();

    public void log(String line) {
        logBuffer.add(line);
    }

    public LogSlice getLogsFrom(int fromIndex) {
        int total = logBuffer.size();
        int safeFrom = Math.max(0, Math.min(fromIndex, total));
        List<String> lines = logBuffer.subList(safeFrom, total);
        return new LogSlice(lines, total);
    }
}
